202constexpr Variant VariantFromProductId(uint32_t product_id) noexcept {
203 switch (product_id & 0xFFFFFF00U) {
204 case 0x06021100U: return Variant::Sfm6000D_50;
205 case 0x06021200U: return Variant::Sfm6000D_20;
206 case 0x06021400U: return Variant::Sfm6000D_5;
207 default: break;
208 }
209 /* SF06: 0x0403 <variant> <revision>. The low byte is a revision — parts
210 * ship as 0x...1x and 0x...01 (an SFM4300-20-P read 0x04030301 on the
211 * 2026-10-02 bench) — so it is ignored. */
212 switch (product_id & 0xFFFFFF00U) {
213 case 0x04030100U: return Variant::Sfm4300_20_B;
214 case 0x04030200U: return Variant::Sfm4300_20_O;
215 case 0x04030300U: return Variant::Sfm4300_20_P;
216 case 0x04030900U: return Variant::Sfm4300_50_B;
217 case 0x04030700U: return Variant::Sfm4300_50_O;
218 case 0x04030600U: return Variant::Sfm4300_50_P;
219 default: return Variant::Unknown;
220 }
221}

249constexpr bool SupportsGas(Variant v, Gas g) noexcept {
250 switch (v) {
251 case Variant::Sfm4300_50_B:
252 case Variant::Sfm4300_50_O:
253 case Variant::Sfm4300_50_P:
254 return g == Gas::O2 || g == Gas::Air || g == Gas::AirO2Mix;
255 case Variant::Sfm4300_20_B:
256 case Variant::Sfm4300_20_O:
257 case Variant::Sfm4300_20_P:
258 return g != Gas::Ar;
259 case Variant::Sfm6000D_50:
260 case Variant::Sfm6000D_20:
261 case Variant::Sfm6000D_5:
262 return g != Gas::N2OO2Mix && g != Gas::CO2O2Mix;
263 case Variant::Unknown:
264 return g == Gas::O2 || g == Gas::Air;
265 }
266 return false;
267}

323constexpr Gas GasFromStatusNibble(Family family, uint8_t nibble) noexcept {
324 if (family == Family::Sfx6000) {
325 switch (nibble & 0x0FU) {
326 case 0x0: return Gas::O2;
327 case 0x1: return Gas::Air;
328 case 0x2: return Gas::CO2;

336 case 0x0: return Gas::O2;

340 case 0x4: return Gas::AirO2Mix;
341 case 0x5: return Gas::N2OO2Mix;
