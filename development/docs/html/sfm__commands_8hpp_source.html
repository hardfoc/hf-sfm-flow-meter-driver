100constexpr uint8_t Crc8(const uint8_t* data, std::size_t len) noexcept {
101 uint8_t crc = 0xFFU;
102 for (std::size_t i = 0; i < len; ++i) {
103 crc ^= data[i];
104 for (int bit = 0; bit < 8; ++bit) {
105 crc = static_cast<uint8_t>((crc & 0x80U) != 0U ? ((crc << 1) ^ 0x31U) : (crc << 1));
106 }
107 }
108 return crc;
109}
