81 DriverResult<ProductInfo> ReadProductIdentifier() noexcept {
82 if (!i2c_.EnsureInitialized()) {
83 return DriverResult<ProductInfo>::failure(DriverError::NotInitialized);
84 }
85 if (!WriteCommand(cmd::kReadProductIdentifier)) {
86 return DriverResult<ProductInfo>::failure(DriverError::BusWrite);
87 }
88 uint16_t words[6]{};
89 const DriverError err = ReadWords(words, 6);
90 if (err != DriverError::None) {
91 return DriverResult<ProductInfo>::failure(err);
92 }
93 ProductInfo info{};
94 info.product_id = (static_cast<uint32_t>(words[0]) << 16) | words[1];
95 for (std::size_t i = 0; i < 4; ++i) {
96 info.serial[2 * i] = static_cast<uint8_t>(words[2 + i] >> 8);
97 info.serial[2 * i + 1] = static_cast<uint8_t>(words[2 + i] & 0xFFU);
98 }
99 info.variant = VariantFromProductId(info.product_id);
100 variant_ = info.variant;
101 if (variant_ != Variant::Unknown) {
102 family_ = FamilyOf(variant_);
103 }
104 return DriverResult<ProductInfo>::success(info);
105 }

113 DriverResult<Scaling> ReadScaleOffsetUnit(Gas g) noexcept {
114 if (!i2c_.EnsureInitialized()) {
115 return DriverResult<Scaling>::failure(DriverError::NotInitialized);
116 }
117 const uint16_t start = StartCommandFor(family_, g);
118 if (start == 0U) {
119 return DriverResult<Scaling>::failure(DriverError::UnsupportedGas);
120 }
121 if (!WriteCommandWithArg(cmd::kReadScaleOffsetUnit, start)) {
122 return DriverResult<Scaling>::failure(DriverError::BusWrite);
123 }
124 /* SFx6000: the parameters sit behind an output-buffer pointer and
125 * include full-scale flow and the SEMI gas ID (reference §3.3.12). */
126 const bool sfx = (family_ == Family::Sfx6000);
127 if (sfx && !WriteCommand(cmd::kSfx6000PointerGasInfo)) {
128 return DriverResult<Scaling>::failure(DriverError::BusWrite);
129 }
130 uint16_t words[5]{};
131 const DriverError err = ReadWords(words, sfx ? 5 : 3);
132 if (err != DriverError::None) {
133 return DriverResult<Scaling>::failure(err);
134 }
135 Scaling s{};
136 s.scale = static_cast<int16_t>(words[0]);
137 s.offset = static_cast<int16_t>(words[1]);
138 s.unit = words[2];
139 if (sfx) {
140 s.full_scale_raw = static_cast<int16_t>(words[3]);
141 s.gas_id = words[4];
142 }
143 s.valid = (s.scale != 0);
144 return DriverResult<Scaling>::success(s);
145 }

157 DriverResult<void> StartContinuous(Gas g, uint16_t o2_permille = 0) noexcept {
158 if (!i2c_.EnsureInitialized()) {
159 return DriverResult<void>::failure(DriverError::NotInitialized);
160 }
161 if (variant_ == Variant::Unknown && GasNeedsKnownFamily(g)) {
162 /* The start code for this gas means another gas on the other
163 * family: read the identifier (or AssumeVariant) first. */
164 return DriverResult<void>::failure(DriverError::VariantUnknown);
165 }
166 if (!SupportsGas(variant_, g)) {
167 return DriverResult<void>::failure(DriverError::UnsupportedGas);
168 }
169 if (IsMixture(g) && o2_permille > 1000U) {
170 return DriverResult<void>::failure(DriverError::InvalidParameter);
171 }
172 const auto sc = ReadScaleOffsetUnit(g);
173 if (!sc.ok()) {
174 return DriverResult<void>::failure(sc.error);
175 }
176 const uint16_t start = StartCommandFor(family_, g);
177 const bool ok = IsMixture(g) ? WriteCommandWithArg(start, o2_permille)
178 : WriteCommand(start);
179 if (!ok) {
180 return DriverResult<void>::failure(DriverError::BusWrite);
181 }
182 scaling_ = sc.value;
183 active_gas_ = g;
184 measuring_ = true;
185 i2c_.DelayMs(timing::kFirstResultAfterStartMs);
186 return DriverResult<void>::success();
187 }

222 static DriverResult<Measurement> DecodeMeasurement(Family family, const Scaling& scaling,
223 const uint8_t frame[9]) noexcept {
224 uint16_t words[3]{};
225 const DriverError err = DecodeWords(frame, 3, words);
226 if (err != DriverError::None) {
227 return DriverResult<Measurement>::failure(err);
228 }
229 Measurement m{};
230 m.flow_raw = static_cast<int16_t>(words[0]);
231 m.status.raw = words[2];
232 m.flow = scaling.valid ? scaling.ToFlow(m.flow_raw) : 0.0f;
233 if (family == Family::Sfx6000) {
234 /* Word 2 is reserved on SFx6000; temperature has its own read. */
235 m.temperature_valid = false;
236 } else {
237 m.temperature_raw = static_cast<int16_t>(words[1]);
238 m.temperature_c = static_cast<float>(m.temperature_raw) / kTemperatureScalePerC;
239 }
240 return DriverResult<Measurement>::success(m);
241 }

248 DriverResult<float> ReadTemperature() noexcept {
249 if (family_ != Family::Sfx6000) {
250 return DriverResult<float>::failure(DriverError::NotSupported);
251 }
252 if (!WriteCommand(cmd::kSfx6000PointerTemperature)) {
253 return DriverResult<float>::failure(DriverError::BusWrite);
254 }
255 uint16_t word[1]{};
256 const DriverError err = ReadWords(word, 1);
257 const bool back = WriteCommand(cmd::kSfx6000PointerResult);
258 if (err != DriverError::None) {
259 return DriverResult<float>::failure(err);
260 }
261 if (!back) {
262 return DriverResult<float>::failure(DriverError::BusWrite);
263 }
264 return DriverResult<float>::success(static_cast<float>(static_cast<int16_t>(word[0])) /
265 kTemperatureScalePerC);
266 }

375 static void EncodeWord(uint16_t word, uint8_t out[3]) noexcept {
376 out[0] = static_cast<uint8_t>(word >> 8);
377 out[1] = static_cast<uint8_t>(word & 0xFFU);
378 out[2] = Crc8(out, 2);
379 }
380
382 static DriverError DecodeWords(const uint8_t* in, std::size_t count, uint16_t* out) noexcept {
383 for (std::size_t i = 0; i < count; ++i) {
384 const uint8_t* p = in + 3 * i;
385 if (Crc8(p, 2) != p[2]) {
386 return DriverError::Crc;
387 }
388 out[i] = static_cast<uint16_t>((static_cast<uint16_t>(p[0]) << 8) | p[1]);
389 }
390 return DriverError::None;
391 }

393private:
394 bool WriteCommand(uint16_t command) noexcept {
395 const uint8_t tx[2] = {static_cast<uint8_t>(command >> 8),
396 static_cast<uint8_t>(command & 0xFFU)};
397 return i2c_.Write(address_, tx, sizeof(tx));
398 }
399
400 bool WriteCommandWithArg(uint16_t command, uint16_t arg) noexcept {
401 uint8_t tx[5];
402 tx[0] = static_cast<uint8_t>(command >> 8);
403 tx[1] = static_cast<uint8_t>(command & 0xFFU);
404 EncodeWord(arg, &tx[2]);
405 return i2c_.Write(address_, tx, sizeof(tx));
406 }

408 DriverError ReadWords(uint16_t* out, std::size_t count) noexcept {
409 if (count == 0 || count > 6) {
410 return DriverError::InvalidParameter;
411 }
412 uint8_t rx[18]{};
413 if (!i2c_.Read(address_, rx, 3 * count)) {
414 return DriverError::NoData;
415 }
416 return DecodeWords(rx, count, out);
417 }
