227constexpr Variant VariantFromProductId(uint32_t product_id) noexcept {
228 switch (product_id & 0xFFFFFF00U) {
229 case 0x06021100U: return Variant::Sfm6000D_50;
230 case 0x06021200U: return Variant::Sfm6000D_20;
231 case 0x06021400U: return Variant::Sfm6000D_5;
232 default: break;
233 }
234 /* SF06: 0x0403 <variant> <revision>. The low byte is a revision — parts
235 * ship as 0x...1x and 0x...01 (an SFM4300-20-P read 0x04030301 on the
236 * 2026-10-02 bench) — so it is ignored. */
237 switch (product_id & 0xFFFFFF00U) {
238 case 0x04030100U: return Variant::Sfm4300_20_B;
239 case 0x04030200U: return Variant::Sfm4300_20_O;
240 case 0x04030300U: return Variant::Sfm4300_20_P;
241 case 0x04030900U: return Variant::Sfm4300_50_B;
242 case 0x04030700U: return Variant::Sfm4300_50_O;
243 case 0x04030600U: return Variant::Sfm4300_50_P;
244 case 0x04020100U: return Variant::Sfm3003_300_CL;
245 case 0x04020700U: return Variant::Sfm3003_300_CE;
246 case 0x04020500U: return Variant::Sfm3013_300_CL;
247 case 0x04020200U: return Variant::Sfm3013_300_CLM;
248 case 0x04020600U: return Variant::Sfm3019_240;

288constexpr bool SupportsGas(Variant v, Gas g) noexcept {
289 switch (v) {
290 case Variant::Sfm4300_50_B:
291 case Variant::Sfm4300_50_O:
292 case Variant::Sfm4300_50_P:
293 case Variant::Sfm3003_300_CL:
294 case Variant::Sfm3003_300_CE:
295 case Variant::Sfm3013_300_CL:
296 case Variant::Sfm3019_240:
297 case Variant::Sfm3119_240_CL:
298 return g == Gas::O2 || g == Gas::Air || g == Gas::AirO2Mix;
299 case Variant::Sfm3013_300_CLM:
300 return g == Gas::O2 || g == Gas::Air || g == Gas::AirO2Mix || g == Gas::HeOx ||
301 g == Gas::HeOxO2Mix;
302 case Variant::Sfm4300_20_B:
303 case Variant::Sfm4300_20_O:
304 case Variant::Sfm4300_20_P:
305 return g != Gas::Ar && g != Gas::HeOx && g != Gas::HeOxO2Mix;
306 case Variant::Sfm6000D_50:
307 case Variant::Sfm6000D_20:
308 case Variant::Sfm6000D_5:
309 return g != Gas::N2OO2Mix && g != Gas::CO2O2Mix && g != Gas::HeOx &&
310 g != Gas::HeOxO2Mix;
311 case Variant::Unknown:
312 return g == Gas::O2 || g == Gas::Air;
313 }
314 return false;

377constexpr bool StatusNibbleIsKnown(Family family, uint8_t nibble) noexcept {
378 const uint8_t n = nibble & 0x0FU;

392constexpr Gas GasFromStatusNibble(Family family, uint8_t nibble) noexcept {
393 if (family == Family::Sfx6000) {
394 switch (nibble & 0x0FU) {
395 case 0x0: return Gas::O2;
396 case 0x1: return Gas::Air;
397 case 0x2: return Gas::CO2;
398 case 0x3: return Gas::N2O;
399 case 0x4: return Gas::Ar;
400 case 0xA: return Gas::AirO2Mix;
401 default: return Gas::Air;
402 }
403 }
404 switch (nibble & 0x0FU) {
405 case 0x0: return Gas::O2;
406 case 0x1: return Gas::Air;
407 case 0x2: return Gas::N2O;
408 case 0x3: return Gas::CO2;
409 case 0x6: return Gas::AirO2Mix;
410 case 0x7: return Gas::N2OO2Mix;
411 case 0x8: return Gas::CO2O2Mix;
412 default: return Gas::Air; // 4, 5 reserved; 9 to 15 unused
413 }
414}
415
416static_assert(GasFromStatusNibble(Family::Sf06, 0x6) == Gas::AirO2Mix,
417 "SFM4300 data sheet Table 5: Air/O2 (0x3632) reports 0b0110");

423constexpr Gas GasFromStatusNibble(Variant variant, uint8_t nibble) noexcept {
424 if (variant == Variant::Sfm3013_300_CLM) {
425 if ((nibble & 0x0FU) == 0x2U) {
426 return Gas::HeOx;
427 }
428 if ((nibble & 0x0FU) == 0x7U) {
429 return Gas::HeOxO2Mix;
430 }
431 }
432 return GasFromStatusNibble(FamilyOf(variant), nibble);
433}
